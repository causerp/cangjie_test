var last : Float64 = 42.0
let A : Float64 = 3877.0
let C : Float64 = 29573.0
let M : Float64 = 139968.0

func rand(m:Float64):Float64{
	last = (last * A + C) % M
	return m * last / M
}

func makeCumulative(table : HashMap<String,Float64>){
    var temp = table.clone()
	var last : ?String = None
	for (c in temp.keys()){
		match (last){
			case Some(V) => temp.put(c, temp[c]+temp[V])
			case _ => 0.0
		}
		last = c
	}
	return temp
}

func fastaRepeat(n_:Int64, seq:String){
	var seqi = 0
	var lenOut = 60
	var n = n_
	while (n > 0) {
	if (n<lenOut) {lenOut = n}
    if (seqi + lenOut < seq.size) {
        print( seq.substring(seqi, lenOut) + "\n" );
        seqi += lenOut;
    } else {
        var s = seq.substring(seqi);
        seqi = lenOut - s.size;
        print( s + seq.substring(0, seqi) + "\n" );
    }	
   	n -= lenOut;
	}
}

func fastaRandom(n_:Int64, table:HashMap<String,Float64> ){
    var temp = makeCumulative(table)
    for (c in temp){
        print("${c[0]} ${c[1]} \n")
    }
	var line = ArrayList<String>(60,{x:Int64=>""})
	var n = n_
	while(n > 0){
		if (n < line.size){line = ArrayList<String>(n,{x:Int64=>""})}
		var x = 0
		while (x < line.size){
			var r = rand(1.0)
			for (c in temp.keys()){
				if (r < temp[c]){
					line.set(x,c)
					break
				}
			}
			x++
		}
		for (i in line){
			print(i)
		}
		print("\n")
		n -= line.size
	}
}

	fastaRandom(3*n, IUB)

	print("\n>THREE Homo sapiens frequency\n")
	fastaRandom(5*n, HomoSap)

}
